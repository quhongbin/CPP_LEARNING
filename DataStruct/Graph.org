** 9.1 图

*** 图的定义

图（Graph）是由顶点（Vertex）集合 V 和边（Edge）集合 E 组成的数据结构，记作 G = (V, E)。

- *顶点*：图中的数据元素
- *边*：顶点之间的关系（连接）

*** 图的分类

| 类型 | 说明 | 边数上限 |
|------+-----------------------------+------------------|
| 无向图 | 边没有方向，(u,v) = (v,u) | n(n-1)/2 |
| 有向图 | 边有方向，<u,v> ≠ <v,u> | n(n-1) |
| 带权图 | 边上有权值（代价/距离） | - |
| 完全图 | 任意两个顶点之间都有边 | n(n-1)/2 |
| 连通图 | 任意两个顶点之间都有路径 | - |
| 强连通图 | 有向图中任意两点互相可达 | - |

*** 图的基本概念

- *度（Degree）*：与顶点相连的边数。有向图分为入度和出度
- *路径（Path）*：从顶点 u 到顶点 v 的顶点序列
- *回路（Cycle）*：起点和终点相同的路径
- *连通分量*：无向图中的极大连通子图
- *强连通分量*：有向图中的极大强连通子图
- *生成树*：包含所有顶点的极小连通子图（n-1 条边）

*** 图的存储结构

**** 1. 邻接矩阵（Adjacency Matrix）

用二维数组表示顶点之间的邻接关系。

#+begin_src cpp
#include <iostream>
#include <vector>
#include <climits>

using namespace std;

const int INF = INT_MAX;

class GraphMatrix {
private:
    int n;                          // 顶点数
    vector<vector<int>> adj;        // 邻接矩阵
    vector<string> vertices;        // 顶点名称

public:
    GraphMatrix(int n) : n(n) {
        adj.resize(n, vector<int>(n, INF));
        // 对角线为0（自身到自身距离为0）
        for (int i = 0; i < n; i++)
            adj[i][i] = 0;
    }

    void addVertex(const string& name) {
        vertices.push_back(name);
    }

    // 无向图：对称添加
    void addEdge(int u, int v, int weight = 1) {
        if (u >= 0 && u < n && v >= 0 && v < n) {
            adj[u][v] = weight;
            adj[v][u] = weight;  // 无向图
        }
    }

    // 有向图
    void addDirectedEdge(int u, int v, int weight = 1) {
        if (u >= 0 && u < n && v >= 0 && v < n) {
            adj[u][v] = weight;
        }
    }

    void print() {
        cout << "  ";
        for (int i = 0; i < n; i++)
            cout << vertices[i] << " ";
        cout << endl;

        for (int i = 0; i < n; i++) {
            cout << vertices[i] << " ";
            for (int j = 0; j < n; j++) {
                if (adj[i][j] == INF)
                    cout << "∞ ";
                else
                    cout << adj[i][j] << " ";
            }
            cout << endl;
        }
    }

    int getEdge(int u, int v) const {
        return adj[u][v];
    }
};

int main() {
    GraphMatrix g(4);
    g.addVertex("A");
    g.addVertex("B");
    g.addVertex("C");
    g.addVertex("D");

    g.addEdge(0, 1, 1);
    g.addEdge(0, 2, 1);
    g.addEdge(1, 3, 1);
    g.addEdge(2, 3, 1);

    g.print();
    return 0;
}
#+end_src

*特点*：
- 空间复杂度：O(n²)
- 查询边的时间复杂度：O(1)
- 适合稠密图

**** 2. 邻接表（Adjacency List）

用链表（或动态数组）存储每个顶点的邻接顶点。

#+begin_src cpp
#include <iostream>
#include <vector>
#include <list>

using namespace std;

class GraphList {
private:
    int n;                              // 顶点数
    vector<list<pair<int, int>>> adj;   // 邻接表 {顶点, 权值}

public:
    GraphList(int n) : n(n) {
        adj.resize(n);
    }

    // 无向图
    void addEdge(int u, int v, int weight = 1) {
        adj[u].push_back({v, weight});
        adj[v].push_back({u, weight});
    }

    // 有向图
    void addDirectedEdge(int u, int v, int weight = 1) {
        adj[u].push_back({v, weight});
    }

    void print() {
        for (int i = 0; i < n; i++) {
            cout << "Vertex " << i << ": ";
            for (auto& edge : adj[i]) {
                cout << "-> " << edge.first
                     << "(w=" << edge.second << ") ";
            }
            cout << endl;
        }
    }

    const list<pair<int, int>>& getNeighbors(int u) const {
        return adj[u];
    }
};

int main() {
    GraphList g(4);

    g.addEdge(0, 1, 1);
    g.addEdge(0, 2, 1);
    g.addEdge(1, 3, 1);
    g.addEdge(2, 3, 1);

    g.print();
    return 0;
}
#+end_src

*特点*：
- 空间复杂度：O(n + e)，e 为边数
- 适合稀疏图
- 查询某顶点的所有邻接点效率高

*** 两种存储结构对比

| 特性 | 邻接矩阵 | 邻接表 |
|------+-----------------+-----------------|
| 空间复杂度 | O(n²) | O(n + e) |
| 查询边是否存在 | O(1) | O(degree) |
| 遍历所有边 | O(n²) | O(n + e) |
| 适合场景 | 稠密图 | 稀疏图 |
| 增加/删除边 | O(1) | O(1) |
| 增加/删除顶点 | O(n²) | O(n + e) |

** 9.2 图基础操作

*** 添加/删除顶点

#+begin_src cpp
// 邻接表实现
class Graph {
private:
    int n;
    vector<list<int>> adj;

public:
    Graph(int n) : n(n) {
        adj.resize(n);
    }

    // 添加顶点（扩展图）
    void addVertex() {
        adj.push_back(list<int>());
        n++;
    }

    // 删除顶点（需要重新编号）
    void removeVertex(int v) {
        if (v < 0 || v >= n) return;

        // 从所有邻接表中删除该顶点
        for (int i = 0; i < n; i++) {
            if (i != v) {
                adj[i].remove(v);
                // 重新编号：大于v的顶点编号减1
                for (auto& neighbor : adj[i]) {
                    if (neighbor > v) neighbor--;
                }
            }
        }

        adj.erase(adj.begin() + v);
        n--;
    }
};

#+begin_src cpp
class Graph {
    // ... 同上 ...

    // 无向图：度数 = 邻接表长度
    int degree(int v) const {
        if (v < 0 || v >= n) return -1;
        return adj[v].size();
    }

    // 有向图：出度
    int outDegree(int v) const {
        if (v < 0 || v >= n) return -1;
        return adj[v].size();
    }

    // 有向图：入度（需要遍历所有邻接表）
    int inDegree(int v) const {
        if (v < 0 || v >= n) return -1;
        int count = 0;
        for (int i = 0; i < n; i++) {
            for (int neighbor : adj[i]) {
                if (neighbor == v) count++;
            }
        }
        return count;
    }
};

** 9.3 图的遍历

*** 深度优先搜索（DFS）

从起始顶点出发，尽可能深地搜索图的分支。

**** 递归实现

#+begin_src cpp
#include <iostream>
#include <vector>
#include <list>

using namespace std;

class Graph {
private:
    int n;
    vector<list<int>> adj;

    void dfsUtil(int v, vector<bool>& visited) {
        visited[v] = true;
        cout << v << " ";

        for (int neighbor : adj[v]) {
            if (!visited[neighbor]) {
                dfsUtil(neighbor, visited);
            }
        }
    }

public:
    Graph(int n) : n(n) {
        adj.resize(n);
    }

    void addEdge(int u, int v) {
        adj[u].push_back(v);
        adj[v].push_back(u);  // 无向图
    }

    void dfs(int start) {
        vector<bool> visited(n, false);
        cout << "DFS from " << start << ": ";
        dfsUtil(start, visited);
        cout << endl;

        // 处理非连通图
        for (int i = 0; i < n; i++) {
            if (!visited[i]) {
                cout << "DFS from " << i << ": ";
                dfsUtil(i, visited);
                cout << endl;
            }
        }
    }
};

int main() {
    Graph g(5);
    g.addEdge(0, 1);
    g.addEdge(0, 2);
    g.addEdge(1, 3);
    g.addEdge(2, 4);

    g.dfs(0);
    // 输出: DFS from 0: 0 1 3 2 4
    return 0;
}
#+end_src

*** 广度优先搜索（BFS）

从起始顶点出发，逐层访问邻接顶点。

#+begin_src cpp
#include <iostream>
#include <vector>
#include <list>
#include <queue>

using namespace std;

class Graph {
private:
    int n;
    vector<list<int>> adj;

public:
    Graph(int n) : n(n) {
        adj.resize(n);
    }

    void addEdge(int u, int v) {
        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    void bfs(int start) {
        vector<bool> visited(n, false);
        queue<int> q;

        q.push(start);
        visited[start] = true;

        cout << "BFS from " << start << ": ";
        while (!q.empty()) {
            int v = q.front();
            q.pop();
            cout << v << " ";

            for (int neighbor : adj[v]) {
                if (!visited[neighbor]) {
                    visited[neighbor] = true;
                    q.push(neighbor);
                }
            }
        }
        cout << endl;
    }

    // BFS 求最短路径（无权图）
    vector<int> shortestPath(int start, int end) {
        vector<bool> visited(n, false);
        vector<int> parent(n, -1);
        queue<int> q;

        q.push(start);
        visited[start] = true;

        while (!q.empty()) {
            int v = q.front();
            q.pop();

            if (v == end) break;

            for (int neighbor : adj[v]) {
                if (!visited[neighbor]) {
                    visited[neighbor] = true;
                    parent[neighbor] = v;
                    q.push(neighbor);
                }
            }
        }

        // 重构路径
        vector<int> path;
        if (visited[end]) {
            for (int v = end; v != -1; v = parent[v]) {
                path.push_back(v);
            }
            reverse(path.begin(), path.end());
        }

        return path;
    }
};

int main() {
    Graph g(6);
    g.addEdge(0, 1);
    g.addEdge(0, 2);
    g.addEdge(1, 3);
    g.addEdge(2, 4);
    g.addEdge(3, 5);
    g.addEdge(4, 5);

    g.bfs(0);
    // 输出: BFS from 0: 0 1 2 3 4 5

    vector<int> path = g.shortestPath(0, 5);
    cout << "Shortest path 0->5: ";
    for (int v : path) cout << v << " ";
    // 输出: Shortest path 0->5: 0 1 3 5

    return 0;
}
#+end_src

**** BFS 应用

- 最短路径（无权图）
- 连通分量
- 层次遍历
- 二分图检测

*** DFS 与 BFS 对比

| 特性 | DFS | BFS |
|------+-----------------+-----------------|
| 数据结构 | 栈（递归/显式） | 队列 |
| 空间复杂度 | O(n) | O(n) |
| 时间复杂度 | O(n + e) | O(n + e) |
| 适用场景 | 路径查找、拓扑排序 | 最短路径、层次遍历 |
| 遍历顺序 | 深度优先 | 广度优先 |
| 找到的路径 | 不一定最短 | 无权图最短 |

** 9.4 小结

*** 核心知识点

1. *图的基本概念*
   - 顶点、边、度、路径、回路
   - 有向图、无向图、带权图
   - 连通图、完全图、生成树

2. *存储结构*
   - 邻接矩阵：适合稠密图，查询快
   - 邻接表：适合稀疏图，节省空间

3. *基本操作*
   - 添加/删除顶点和边
   - 获取度数、邻接顶点
   - 检查边是否存在

4. *图遍历*
   - DFS：深度优先，用栈实现
   - BFS：广度优先，用队列实现
   - 两者时间复杂度均为 O(n + e)

*** 复杂度总结

| 操作 | 邻接矩阵 | 邻接表 |
|------+-----------------+-----------------|
| 添加边 | O(1) | O(1) |
| 删除边 | O(1) | O(degree) |
| 查询边 | O(1) | O(degree) |
| 遍历图 | O(n²) | O(n + e) |
| DFS/BFS | O(n²) | O(n + e) |

*** 应用场景

- *社交网络*：好友关系、推荐系统
- *地图导航*：最短路径（Dijkstra、A*）
- *网络路由*：数据包转发
- *任务调度*：拓扑排序
- *网页爬虫*：BFS 遍历链接
- *编译器*：依赖分析、控制流图
